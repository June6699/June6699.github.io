{{- $destination := .Destination -}} {{- $parsedDestination := urls.Parse $destination -}} {{- $src := $destination -}} {{- $sourcePath := "" -}} {{- $imageWidth := 0 -}} {{- $imageHeight := 0 -}} {{- /* 动态优先：./xxx → /images/moments/xxx（源在 content/moments） */ -}} {{- if and (eq .Page.Section "moments") (hasPrefix $src "./") -}} {{- $relativePath := strings.TrimPrefix "./" $parsedDestination.Path -}} {{- $sourcePath = path.Join "content" "moments" $relativePath -}} {{- $src = printf "/images/moments/%s" (strings.TrimPrefix "./" $src) -}} {{- /* 文章：./images/... → /images/...（源在 content/posts/images） */ -}} {{- else if hasPrefix $src "./images/" -}} {{- $relativePath := strings.TrimPrefix "./images/" $parsedDestination.Path -}} {{- $sourcePath = path.Join "content" "posts" "images" $relativePath -}} {{- $src = replace $src "./images/" "/images/" -}} {{- end -}} {{- if and $sourcePath (os.FileExists $sourcePath) -}} {{- $imageConfig := images.Config $sourcePath -}} {{- $imageWidth = $imageConfig.Width -}} {{- $imageHeight = $imageConfig.Height -}} {{- end -}}
